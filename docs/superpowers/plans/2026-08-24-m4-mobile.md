# M4 移动端复刻 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 交付 Android APK 的四屏链路（侧栏首屏 → 任务子列表 → 详情全屏 → 底部抽屉表单），观感逐像素对齐 `docs/05_UI复刻规格-移动端.md`，并落地 R1（SQLCipher）/R2（移动通知）两项技术决策记录。

**Architecture:** 单一 React 前端按平台分叉：`src/lib/platform.ts` 探测 UA 后选择桌面 router（现状不动）或移动 hash-router；移动专属组件放 `src/features/todo/mobile/` 与 `src/components/mobile/`。Rust 壳用 `cfg(desktop)` 门控窗口/Mica/window-state/keyring 能力，`orbit_core` 业务零改动。玻璃组件库自研（不引 vaul），弹簧参数照抄规格。

**Tech Stack:** Tauri 2 Android（aarch64-linux-android）、React 19 + Tailwind v4、react-router v7（移动 createHashRouter）、@dnd-kit（触摸排序）、vitest（纯逻辑单测）、Material Symbols Rounded 自托管字体。

## Global Constraints

以下为全计划约束，每个任务隐式包含（数值从文档逐字抄录，不得改写）：

- 全部界面文案定死中文，不做 i18n（01 文档 §二）。

```ts
/**
 * 平台探测：同一份前端产物跑在桌面 WebView 与移动 WebView 上，
 * 以 UA 判定分叉（Tauri Android WebView UA 含 Android；iOS WKWebView 含 iPhone/iPad）。
 */
export function isMobileUA(ua: string): boolean {
  return /Android|iPhone|iPad|iPod/i.test(ua);
}

export function isMobilePlatform(): boolean {
  return typeof navigator !== "undefined" && isMobileUA(navigator.userAgent);
}

/** Android 手势条兜底高度（05 §五：env(safe-area-inset-bottom) 为 0 时兜底 48px） */
```

```ts
/** Material Symbols 字体就绪探测（ligature 渲染依赖；失败时组件降级为占位圆点） */
export function useFontReady(): boolean {
  const [ready, setReady] = useState(cached ?? false);
  useEffect(() => {
    if (cached != null) return;
    let alive = true;
    document.fonts
      .load('24px "Material Symbols Rounded"', "search")
      .then(() => document.fonts.check('24px "Material Symbols Rounded"'))
      .then((ok) => {
        cached = ok;
        if (alive) setReady(ok);
      })
      .catch(() => {
        cached = false;
        if (alive) setReady(false);
      });
    return () => {
      alive = false;
    };
  }, []);
  return ready || !!cached;
}
```

```tsx
import { useEffect, useRef } from "react";

/**
 * EqSpinner —— 移植 wait-home bounce-blocks 物理模型（05 §五）：
 * 重力 2600 px/s²、restitution 0.62、挤压弹簧 170/14、三球延迟 0/.65/1.3s。
 * rAF 驱动，transform 直写，不触发 React 重渲。
 */
const GRAVITY = 2600;
const RESTITUTION = 0.62;
const SQUASH_STIFFNESS = 170;
const SQUASH_DAMPING = 14;
const DELAYS = [0, 650, 1300];

interface BallState { y: number; vy: number; squash: number; squashV: number; }

export function EqSpinner({ size = 48, color = "#4E8CFF" }: { size?: number; color?: string }) {
  const ref = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const container = ref.current;
    if (!container) return;
    const balls = Array.from(container.children) as HTMLElement[];
    const dropH = size * 0.9;
    const state: BallState[] = balls.map(() => ({ y: -dropH, vy: 0, squash: 1, squashV: 0 }));
    let raf = 0;
    let last = performance.now();
    const tick = (now: number) => {
      const dt = Math.min((now - last) / 1000, 0.032);
      last = now;
      balls.forEach((ball, i) => {
        const s = state[i];
        const active = now >= DELAYS[i]; // 三球延迟入场
        if (active) {
          s.vy += GRAVITY * dt;
          s.y += s.vy * dt * (size / 48); // 尺寸缩放位移
          if (s.y >= 0) {                 // 触地
            s.y = 0;
            s.vy = Math.abs(s.vy) > 40 ? -Math.abs(s.vy) * RESTITUTION : 0;
            s.squashV -= 2600 * dt;       // 挤压冲量
          }
          // 挤压弹簧回弹
          const force = -SQUASH_STIFFNESS * (s.squash - 1);
          s.squashV += force * dt;
          s.squashV -= SQUASH_DAMPING * s.squashV * dt;
          s.squash += s.squashV * dt * 10;
        }
        ball.style.transform = `translateY(${s.y}px) scale(${2 - s.squash}, ${s.squash})`;
        ball.style.opacity = active ? "1" : "0";
      });
      raf = requestAnimationFrame(tick);
    };
    raf = requestAnimationFrame(tick);
    return () => cancelAnimationFrame(raf);
  }, [size]);

  const ballSize = Math.max(size * 0.16, 4);
  return (
    <div ref={ref} className="relative overflow-hidden" style={{ width: size, height: size }}>
      {[0, 1, 2].map((i) => (
        <span key={i} className="absolute rounded-full" style={{ width: ballSize * 2, height: ballSize * 2, left: `${8 + i * 34}%`, top: size - ballSize * 2, background: color, transformOrigin: "center bottom", willChange: "transform" }} />
      ))}
    </div>
  );
}
```

```ts
/** touchend 选点：速度主导(|v|>500px/s 按方向取相邻)，否则最近邻 */
export function pickSnapIndex(offsetPx: number, velocityPxPerS: number, snapsPx: number[]): number {
  const sorted = [...snapsPx].sort((a, b) => a - b);
  if (velocityPxPerS < -500) {
    const uppers = sorted.filter((s) => s < offsetPx);
    if (uppers.length) return snapsPx.indexOf(uppers[uppers.length - 1]);
  }
  if (velocityPxPerS > 500) {
    const lowers = sorted.filter((s) => s > offsetPx);
    if (lowers.length) return snapsPx.indexOf(lowers[0]);
  }
  let best = 0;
  for (let i = 1; i < snapsPx.length; i++) {
    if (Math.abs(snapsPx[i] - offsetPx) < Math.abs(snapsPx[best] - offsetPx)) best = i;
  }
  return best;
}

/** 半隐式欧拉弹簧积分（scrollSpring 1,170,24 / drawerSpring 1,300,25 通用） */
export function springStep(pos: number, vel: number, target: number, stiffness: number, damping: number, dt: number): [number, number] {
  const nv = vel + (-stiffness * (pos - target) - damping * vel) * dt;
  return [pos + nv * dt, nv];
}
```
